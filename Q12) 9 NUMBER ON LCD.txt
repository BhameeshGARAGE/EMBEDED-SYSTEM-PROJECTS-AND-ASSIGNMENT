ADC.C

#include "adc.h"

// Configure button pin using AVR Registers
void ADC_Init(void) {
	BUTTON_DDR &= ~(1 << BUTTON_PIN); // Configure PD7 as input pin
	BUTTON_PORT |= (1 << BUTTON_PIN);  // Enable internal pull-up
}

// Read the physical state of the button
uint8_t ADC_ReadButton(void) {
	if ((BUTTON_PIN_REG & (1 << BUTTON_PIN)) == 0) {
		return BUTTON_PRESSED;
		} else {
		return BUTTON_RELEASED;
	}
}

MAIN.C

#include <avr/io.h>
#include <util/delay.h>
#include <stdint.h>
#include "adc.h"
#include "lcd.h"

// Global execution variables
uint32_t counter = 0;
uint32_t buttonPressMs = 0;
uint8_t isButtonPressed = 0;
uint8_t longPressDetected = 0;

void checkButtonState(void);

int main(void) {
	ADC_Init();
	LCD_Init();
	LCD_UpdateDisplay(counter);

	while (1) {
		checkButtonState();
		_delay_ms(10);
		
		if (isButtonPressed && !longPressDetected) {
			buttonPressMs += 10;
		}
	}
}

void checkButtonState(void) {
	uint8_t buttonState = ADC_ReadButton();

	if (buttonState == BUTTON_PRESSED) {
		if (!isButtonPressed) {
			isButtonPressed = 1;
			buttonPressMs = 0;
			longPressDetected = 0;
		}

		if (!longPressDetected && (buttonPressMs >= 2000)) {
			counter = 0;
			LCD_UpdateDisplay(counter);
			longPressDetected = 1;
		}
	}
	else {
		if (isButtonPressed) {
			if (!longPressDetected) {
				counter++;
				if (counter > 999999999) {
					counter = 0;
				}
				LCD_UpdateDisplay(counter);
			}
			isButtonPressed = 0;
		}
	}
}
